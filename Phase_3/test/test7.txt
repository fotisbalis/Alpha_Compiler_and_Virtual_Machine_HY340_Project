// ===== GLOBAL VARIABLES =====
x = 5;
y = 10;

function globalFunc(a, b) {
    local temp;
    temp = a + b;

    function innerFunc(c) {
        local z;
        z = c + temp;      // should see temp (same function scope boundary)
        return z;
    }

    temp = innerFunc(3);

    if (temp > 0) {
        local y;
        y = temp * 2;
    } else {
        y = temp / 2;
    }

    while (temp < 50) {
        temp++;
        if (temp == 20) {
            break;
        }
    }

    return temp;
}

// ===== CALLS =====
result = globalFunc(x, y);

// ===== SHADOWING TEST =====
function shadowTest() {
    local x;
    x = 100;

    {
        local x;
        x = 200;
    }

    return x;
}

// ===== GLOBAL ACCESS TEST =====
function globalAccess() {
    local x;
    x = 1;
    ::x = 99;          // modify global x
}
