function factorial(a) {
	if(a < 0) return false;	

	if(a == 0) return 1;

	return a * factorial(a - 1);
}

function fibbonacci(a) {
        if(a < 0) return false;

	if(a == 0 or a == 1) return a;

	return fibbonacci(a - 1) + fibbonacci(a - 2);
}

for(i = -1; i <= 10; i++) {
	print("factorial of ", i, " = ", factorial(i), "\n");
	print("fibbonacci of ", i, " = ", fibbonacci(i), "\n\n");
}
