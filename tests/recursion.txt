function factorial(a) {
	if(a < 0) return false;	

	if(a == 0) return 1;

	return a * factorial(a - 1);
}

function fibbonacci(a) {
        if(a < 0) return false;

	if(a == 0 or a == 1) return a;

	return fibbonacci(a - 1) + fibbonacci(a - 2);
}

print("Give a positive integer: ");
x = input();
a = x;
b = x;
print("fibbonacci of ", x, " = ", fibbonacci(a), "\n");
print("factorial of ", x, " = ", factorial(b), "\n");
