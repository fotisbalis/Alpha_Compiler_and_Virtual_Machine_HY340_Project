function isPrime(n) {
	if(n < 2) return false;

	count = 0;
	for(i = 1; i < n; i++) {
		if(n % i == 0)
			count++;
	}

	if(count > 1)
		return false;

	return true;
}

while(true) {
	print("Give a number: ");
	
	if(n = input()) {
		print(n, " is ");
		if(isPrime(n)) 
			println("prime.");
		else 
			println("not prime.");
	}
	else
		break;
}
